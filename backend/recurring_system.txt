from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, update
from sqlalchemy.orm import selectinload

from datetime import datetime
from typing import Optional

from app.models import RecurringTaskLog, User
from app.models.RecurringTaskModel import RecurringTask
from app.models.RecurringTaskStepModel import RecurringTaskStep
from app.db.enums import GlobalRole
from app.core.permissions import AccessService
class RecurringTaskRepository:
    @staticmethod
    def get_base_query(current_user: User, include_deleted: bool = False):
        query = select(RecurringTask).options(
            selectinload(RecurringTask.steps.and_(RecurringTaskStep.deleted_at.is_(None)))
        )

async def execute_recurring_tasks_automation(db: AsyncSession):
    now_utc = datetime.utcnow()
    today = now_utc.date()
    current_time = now_utc.time()
    
    # جلب القوالب النشطة مع خطواتها غير المحذوفة
    stmt = select(RecurringTask).options(
        selectinload(RecurringTask.steps.and_(RecurringTaskStep.deleted_at.is_(None)))
    ).where(
        and_(
            RecurringTask.is_active == True,
            RecurringTask.deleted_at == None,
            RecurringTask.next_run_date <= today
        )
    )
    
    result = await db.execute(stmt)
    templates = result.scalars().all()
    
    executed_count = 0
    for template in templates:
        if template.next_run_date == today and template.run_time > current_time:
            continue
            
        log_entry = RecurringTaskLog(recurring_task_id=template.id, run_at=now_utc)
        
        try:
            task_priority = template.priority.upper() if template.priority else "MEDIUM"
            
            # 1. إنشاء المهمة الرئيسية
            new_task = Task(
                title=template.title,
                description=template.description,
                department_id=template.department_id,
                priority=task_priority,
                status="not_started",
                created_by=template.created_by,
                created_at=now_utc
            )
            db.add(new_task)
            await db.flush()  # للحصول على new_task.id
            
            # 2. نسخ خطوات القالب وإدراجها كـ TaskSteps فعلية للمهمة
            for t_step in template.steps:
                task_step = TaskStep(
                    task_id=new_task.id,
                    title=t_step.title,
                    description=t_step.description,
                    step_order=t_step.step_order,
                    is_parallel=t_step.is_parallel,
                    status="pending",
                    assigned_department_id=t_step.assigned_department_id,
                    assigned_user_id=t_step.assigned_user_id,
                    created_at=now_utc
                )
                db.add(task_step)

            # 3. إرسال الإشعارات
            user_stmt = select(User).where(User.department_id == template.department_id)
            user_result = await db.execute(user_stmt)
            department_users = user_result.scalars().all()
            
            for user in department_users:
                await NotificationService.create(
                    db=db,
                    user_id=user.id,
                    notification_type="task_assigned",
                    title="📅 مهمة دورية تلقائية",
                    body=f"قام النظام بإصدار مهمة دورية آلياً لقسمكم بعنوان: ({template.title}).",
                    related_task_id=new_task.id,
                    extra_data={"automation": True, "template_id": template.id},
                    send_ws=True
                )
            
            # 4. حساب تاريخ التشغيل القادم
            template.next_run_date = calculate_next_run_date(template.next_run_date, template.recurrence_pattern, template.interval_value)
            template.updated_at = now_utc
            
            log_entry.status = "success"
            log_entry.generated_task_id = new_task.id
            executed_count += 1
            
        except Exception as e:
            await db.rollback()
            log_entry.status = "failed"
            log_entry.error_message = str(e)
            
        db.add(log_entry)
        
    await db.commit()
    return {"status": "completed", "tasks_generated": executed_count}


def calculate_next_run_date(current_date: date, pattern: str, interval: int) -> date:
    from datetime import timedelta
    if pattern == "daily":
        return current_date + timedelta(days=interval)
    elif pattern == "weekly":
        return current_date + timedelta(weeks=interval)
    elif pattern == "monthly":
        import calendar
        month = current_date.month - 1 + interval
        year = current_date.year + month // 12
        month = month % 12 + 1
        day = min(current_date.day, calendar.monthrange(year, month)[1])
        return date(year, month, day)
    return current_date + timedelta(days=1)

    @staticmethod
    async def _apply_steps_diff(db: AsyncSession, template: RecurringTask, steps_payload: list):
        """دالة مساعدة لمعالجة التحديث الجزئي (Diff Update) لخطوات القالب مع التحقق المستبق"""
        
        # التحقق المستبق من صحة الأقسام والمستخدمين الممررين في الـ Payload
        await RecurringTaskService._validate_steps_entities(db, steps_payload)

        existing_steps_map = {step.id: step for step in template.steps if step.deleted_at is None}
        payload_step_ids = {s["id"] for s in steps_payload if s.get("id") is not None}
        now_utc = get_utc_now()

        # 1. الحذف الناعم للخطوات غير الموجودة في الـ Payload
        for step_id, step_obj in existing_steps_map.items():
            if step_id not in payload_step_ids:
                step_obj.deleted_at = now_utc

        # 2. التحديث والإضافة
        for step_data in steps_payload:
            step_id = step_data.get("id")
            if step_id and step_id in existing_steps_map:
                # تحديث خطوة موجودة
                step_obj = existing_steps_map[step_id]
                for key, val in step_data.items():
                    if key != "id" and val is not None:
                        setattr(step_obj, key, val)
            elif not step_id:
                # إنشاء خطوة جديدة
                step_data.pop("id", None)
                new_step = RecurringTaskStep(**step_data)
                template.steps.append(new_step)

            return

        dept_ids = {s.get("assigned_department_id") for s in steps_data if s.get("assigned_department_id")}
        user_ids = {s.get("assigned_user_id") for s in steps_data if s.get("assigned_user_id")}

        if dept_ids:
            stmt = select(Department.id).where(Department.id.in_(dept_ids))
            res = await db.execute(stmt)
            existing_dept_ids = set(res.scalars().all())
            missing_depts = dept_ids - existing_dept_ids
            if missing_depts:
                raise HTTPException(
                    status_code=400,
                    detail=f"الأقسام ذات المعرفات التالية غير موجودة في النظام: {list(missing_depts)}"
                )

        # 2. التحقق من المستخدمين
        if user_ids:
            stmt = select(User.id).where(User.id.in_(user_ids))
            res = await db.execute(stmt)
            existing_user_ids = set(res.scalars().all())
            missing_users = user_ids - existing_user_ids

                raise HTTPException(
                    status_code=400,

router = APIRouter(prefix="/applicants", tags=["Applicants"])


@router.get("", response_model=List[ApplicantWithJobOut])
def list_applicants(
    job_model_id: Optional[str] = None,
    db: OrmSession = Depends(get_db),
    admin=Depends(get_current_admin),
):
    filters = {"job_model_id": job_model_id} if job_model_id else {}
    applicants = ApplicantRepository(db).get_all(limit=5000, **filters)
    jm_repo = JobModelRepository(db)

    result = []
    for a in applicants:
        jm = jm_repo.get(a.job_model_id)
        item = ApplicantWithJobOut.model_validate(a)
        item.job_model_name = jm.name if jm else None
        result.append(item)
    return result


@router.post("", response_model=ApplicantCreatedOut)
def create_applicant(payload: ApplicantCreate, db: OrmSession = Depends(get_db), admin=Depends(get_current_admin)):
    repo = ApplicantRepository(db)
    if repo.get_by_national_id(payload.national_id):
        raise HTTPException(status_code=400, detail="الرقم القومي مسجل بالفعل")

    access_code = None
    for _ in range(10):
        candidate = generate_access_code()
        if not repo.get_by(access_code=candidate):
            access_code = candidate
            break
    if not access_code:
        raise HTTPException(status_code=500, detail="تعذر توليد كود دخول فريد، حاول مرة أخرى")

    applicant = repo.create({
        "national_id": payload.national_id,
        "full_name": payload.full_name,
        "phone": payload.phone,
        "job_model_id": payload.job_model_id,
        "access_code": access_code,
        "status": "pending",
    })
    return ApplicantCreatedOut(id=applicant.id, access_code=applicant.access_code)

from typing import Optional

from fastapi import APIRouter, Cookie, Depends, Header, HTTPException, Response, status
from sqlalchemy.orm import Session as OrmSession

from core.config import settings
from core.deps import get_current_admin
from db.session import get_db
from schemas.admin import AdminLogin, AdminLoginResponse, AdminOut
from services.auth_service import AuthService

router = APIRouter(prefix="/auth", tags=["Auth"])


@router.post("/login", response_model=AdminLoginResponse)
def login(payload: AdminLogin, response: Response, db: OrmSession = Depends(get_db)):
    admin, token = AuthService(db).login_admin(payload.username, payload.password)
    if not admin:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="اسم المستخدم أو كلمة المرور غير صحيحة")

    response.set_cookie(
        key="admin_session",
        value=token,
        httponly=True,
        samesite="lax",
        max_age=settings.SESSION_EXPIRE_HOURS * 3600,
    )
    return AdminLoginResponse(token=token, admin=AdminOut.model_validate(admin))


@router.post("/logout")
def logout(
    response: Response,
    db: OrmSession = Depends(get_db),
    authorization: Optional[str] = Header(default=None),
    admin_session: Optional[str] = Cookie(default=None),
):
    token = None
    if authorization and authorization.lower().startswith("bearer "):
        token = authorization[7:].strip()
    else:
        token = admin_session

    if token:
        AuthService(db).logout(token)

    response.delete_cookie("admin_session")
    return {"success": True}


@router.get("/me", response_model=AdminOut)
def me(admin=Depends(get_current_admin)):
    return AdminOut.model_validate(admin)

@router.post("/login")
def exam_login(payload: ExamLoginRequest, response: Response, db: OrmSession = Depends(get_db)):
    applicant_repo = ApplicantRepository(db)
    applicant = applicant_repo.get_by(access_code=payload.access_code.strip().upper())
    if not applicant:
        raise HTTPException(status_code=404, detail="كود الدخول غير صحيح")
    if applicant.status == "finished":
        raise HTTPException(status_code=403, detail="لقد قمت بتأدية هذا الاختبار من قبل")

    job_model = JobModelRepository(db).get(applicant.job_model_id)

    if applicant.status == "pending":
        applicant.status = "in_progress"
    if not applicant.exam_started_at:
        applicant.exam_started_at = datetime.utcnow()
    db.commit()
    db.refresh(applicant)

    token = AuthService(db).create_applicant_session(applicant.id)
    response.set_cookie(
        key="applicant_session",
        value=token,
        httponly=True,
        samesite="lax",
        max_age=settings.SESSION_EXPIRE_HOURS * 3600,
    )

    questions = QuestionRepository(db).list_by_job_model(applicant.job_model_id)
    saved_responses = {r.question_id: r for r in ResponseRepository(db).list_by_user(applicant.id)}

    questions_payload = []
    for q in questions:
        data = json.loads(q.question_data)
        safe_data = {k: v for k, v in data.items() if k not in ("correct_index", "answer_keywords")}
        saved = saved_responses.get(q.id)
        questions_payload.append({
            "id": q.id,
            "q_type": q.q_type,
            "question_text": q.question_text,
            "question_data": safe_data,
            "image_path": q.image_path,
            "points": q.points,
            "saved_answer": json.loads(saved.answer_data) if saved and saved.answer_data else None,
        })

    return {
        "token": token,
        "user": {
            "id": applicant.id,
            "full_name": applicant.full_name,
            "job_model_name": job_model.name if job_model else "",
            "duration_minutes": job_model.duration_minutes if job_model else 60,
            "exam_started_at": applicant.exam_started_at,
        },
        "questions": questions_payload,
    }

router = APIRouter(prefix="/questions", tags=["Questions"])

ALLOWED_IMAGE_EXTENSIONS = {"png", "jpg", "jpeg", "gif", "webp"}


def _to_out(q) -> QuestionOut:
    return QuestionOut(
        id=q.id,
        job_model_id=q.job_model_id,
        q_type=q.q_type,
        question_text=q.question_text,
        question_data=json.loads(q.question_data),
        image_path=q.image_path,
        points=q.points,
        order_index=q.order_index,
    )


@router.get("", response_model=List[QuestionOut])
def list_questions(job_model_id: str, db: OrmSession = Depends(get_db), admin=Depends(get_current_admin)):
    questions = QuestionRepository(db).list_by_job_model(job_model_id)
    return [_to_out(q) for q in questions]


@router.post("", response_model=QuestionOut)
def create_question(payload: QuestionCreate, db: OrmSession = Depends(get_db), admin=Depends(get_current_admin)):
    q = QuestionRepository(db).create({
        "job_model_id": payload.job_model_id,
        "q_type": payload.q_type,
        "question_text": payload.question_text,
        "question_data": json.dumps(payload.question_data, ensure_ascii=False),
        "points": payload.points,
        "order_index": payload.order_index,
        "image_path": payload.image_path,
    })
    return _to_out(q)


@router.put("/{question_id}", response_model=QuestionOut)
def update_question(
    question_id: str,
    payload: QuestionUpdate,
    db: OrmSession = Depends(get_db),
    admin=Depends(get_current_admin),
):
    repo = QuestionRepository(db)
    obj = repo.get(question_id)
    if not obj:
        raise HTTPException(status_code=404, detail="السؤال غير موجود")

    update_data = payload.model_dump(exclude_unset=True)
    if update_data.get("question_data") is not None:
        update_data["question_data"] = json.dumps(update_data["question_data"], ensure_ascii=False)

    obj = repo.update(obj, update_data)
    return _to_out(obj)


@router.delete("/{question_id}")
def delete_question(question_id: str, db: OrmSession = Depends(get_db), admin=Depends(get_current_admin)):
    repo = QuestionRepository(db)
    if not repo.get(question_id):
        raise HTTPException(status_code=404, detail="السؤال غير موجود")

    ResponseRepository(db).delete_by_question(question_id)
    repo.delete(question_id)
    return {"success": True}


@router.post("/upload-image")
def upload_image(file: UploadFile = File(...), admin=Depends(get_current_admin)):
    ext = file.filename.rsplit(".", 1)[-1].lower() if file.filename and "." in file.filename else ""
    if ext not in ALLOWED_IMAGE_EXTENSIONS:
        raise HTTPException(status_code=400, detail="صيغة الصورة غير مدعومة")

    os.makedirs(settings.UPLOADS_DIR, exist_ok=True)
    unique_name = f"{uuid.uuid4().hex}.{ext}"
    dest_path = os.path.join(settings.UPLOADS_DIR, unique_name)
    with open(dest_path, "wb") as f:
        f.write(file.file.read())

    return {"path": f"/uploads/{unique_name}"}

"""
أدوات الأمان: تجزئة كلمات المرور (PBKDF2-HMAC-SHA256 عبر hashlib القياسية،
بدون الاعتماد على مكتبات خارجية تحتاج بناء/ترجمة)، وتوليد رموز الجلسات
(Session Tokens) وأكواد دخول المتقدمين.
"""

import hashlib
import hmac
import os
import secrets
import string
from datetime import datetime, timedelta
from typing import Optional, Any
from jose import JWTError, jwt
from core.config import settings

PBKDF2_ITERATIONS = 260_000


def _derive(password: str, salt: bytes, iterations: int = PBKDF2_ITERATIONS) -> bytes:
    return hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, iterations)


def hash_password(password: str) -> str:
    """إرجاع تجزئة بصيغة 'salt_hex$digest_hex' لتخزينها في قاعدة البيانات."""
    salt = os.urandom(16)
    digest = _derive(password, salt)
    return f"{salt.hex()}${digest.hex()}"


def verify_password(password: str, password_hash: str) -> bool:
    try:
        salt_hex, digest_hex = password_hash.split("$")
    except (ValueError, AttributeError):
        return False
    salt = bytes.fromhex(salt_hex)
    expected = bytes.fromhex(digest_hex)
    actual = _derive(password, salt)
    return hmac.compare_digest(expected, actual)


def generate_token(length: int = 48) -> str:
    """توليد رمز جلسة عشوائي آمن (Bearer Token)."""
    return secrets.token_urlsafe(length)

def create_access_token(data: dict, expires_delta: Optional[timedelta] = None) -> str:
    to_encode = data.copy()
    expire = datetime.utcnow() + (
        expires_delta or timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    )
    to_encode.update({"exp": expire, "type": "access"})
    return jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)

def generate_access_code(length: int = 8) -> str:
    """توليد كود دخول قصير للمتقدمين (أحرف كبيرة وأرقام)."""
    chars = string.ascii_uppercase + string.digits
    return "".join(secrets.choice(chars) for _ in range(length))

def _set_sqlite_pragma(dbapi_connection, connection_record):
    cursor = dbapi_connection.cursor()
    cursor.execute("PRAGMA journal_mode=WAL;")

    cursor.execute("PRAGMA busy_timeout=30000;")

def get_registered_fonts():
    """تسجيل الخط العربي مرة واحدة فقط، مع الرجوع لـ Helvetica إن لم يوجد ملف الخط."""
    global _fonts_registered, _REGULAR_FONT_NAME, _BOLD_FONT_NAME

        _BOLD_FONT_NAME = "ArabicFont"

    _fonts_registered = True
    return _REGULAR_FONT_NAME, _BOLD_FONT_NAME


def ar(text) -> str:
    """تشكيل النص العربي وضبط اتجاهه ليظهر بشكل صحيح داخل ملف PDF."""

class NumberedCanvas(canvas.Canvas):
    """كلاس وارث من canvas.Canvas يضيف ترقيم صفحات تلقائي 'صفحة X من Y'."""

    def __init__(self, *args, **kwargs):
        canvas.Canvas.__init__(self, *args, **kwargs)
        self._saved_page_states = []

    def showPage(self):
        self._saved_page_states.append(dict(self.__dict__))
        self._startPage()

    def save(self):
        num_pages = len(self._saved_page_states)
        for state in self._saved_page_states:
            self.__dict__.update(state)
            self._draw_page_number(num_pages)
            canvas.Canvas.showPage(self)
        canvas.Canvas.save(self)

    def _draw_page_number(self, total_pages):
        regular_font, _ = get_registered_fonts()
        self.setFont(regular_font, 9)
        self.setFillColor(colors.HexColor("#666666"))
        text = ar(f"صفحة {self._pageNumber} من {total_pages}")
        self.drawCentredString(PAGE_WIDTH / 2, 1.2 * cm, text)
        self.setStrokeColor(colors.HexColor("#cccccc"))
        self.line(2 * cm, 1.6 * cm, PAGE_WIDTH - 2 * cm, 1.6 * cm)

"""
مستودع أساسي عام (Generic Base Repository) يوفر عمليات CRUD القياسية
لأي نموذج SQLAlchemy، لضمان الاعتماد الموحّد على نمط الـ CRUD في كل
أنحاء المشروع كما هو متفق عليه.
"""

from typing import Any, Dict, Generic, List, Optional, Type, TypeVar

from sqlalchemy.orm import Session as OrmSession

ModelType = TypeVar("ModelType")


class BaseRepository(Generic[ModelType]):
    def __init__(self, model: Type[ModelType], db: OrmSession):
        self.model = model
        self.db = db

    def get(self, id: str) -> Optional[ModelType]:
        return self.db.query(self.model).filter(self.model.id == id).first()

    def get_by(self, **filters) -> Optional[ModelType]:
        query = self.db.query(self.model)
        for key, value in filters.items():
            query = query.filter(getattr(self.model, key) == value)
        return query.first()

    def get_all(self, skip: int = 0, limit: int = 1000, **filters) -> List[ModelType]:
        query = self.db.query(self.model)
        for key, value in filters.items():
            if value is not None:
                query = query.filter(getattr(self.model, key) == value)
        return query.offset(skip).limit(limit).all()

    def count(self, **filters) -> int:
        query = self.db.query(self.model)
        for key, value in filters.items():
            if value is not None:
                query = query.filter(getattr(self.model, key) == value)
        return query.count()

    def create(self, obj_in: Dict[str, Any]) -> ModelType:
        db_obj = self.model(**obj_in)
        self.db.add(db_obj)
        self.db.commit()
        self.db.refresh(db_obj)
        return db_obj

    def update(self, db_obj: ModelType, obj_in: Dict[str, Any]) -> ModelType:
        for key, value in obj_in.items():
            if value is not None:
                setattr(db_obj, key, value)
        self.db.commit()
        self.db.refresh(db_obj)
        return db_obj

    def delete(self, id: str) -> bool:
        db_obj = self.get(id)
        if not db_obj:
            return False
        self.db.delete(db_obj)
        self.db.commit()
        return True

        self.db.commit()

    def upsert(self, user_id: str, question_id: str, answer_data, is_correct: bool, score: float):
        """حفظ أو تحديث إجابة المتقدم (UPSERT) - يتعامل مع إعادة الإرسال بأمان."""
        existing = self.get_by_user_and_question(user_id, question_id)
        if existing:
            existing.answer_data = answer_data
            existing.is_correct = int(bool(is_correct))
            existing.score = score
            existing.answered_at = datetime.utcnow()
            self.db.commit()
            self.db.refresh(existing)
            return existing
        return self.create({
            "user_id": user_id,
            "question_id": question_id,
            "answer_data": answer_data,
            "is_correct": int(bool(is_correct)),
            "score": score,
        })

"""
منطق تصحيح الاختبارات وحساب النتائج النهائية واسترجاعها.
"""

import json
from datetime import datetime

from sqlalchemy.orm import Session as OrmSession

from helpers.grading_helpers import normalize_text
from repositories.applicant_repository import ApplicantRepository
from repositories.job_model_repository import JobModelRepository
from repositories.question_repository import QuestionRepository
from repositories.response_repository import ResponseRepository
from repositories.violation_repository import ViolationRepository


class ExamService:
    def __init__(self, db: OrmSession):
        self.db = db
        self.question_repo = QuestionRepository(db)
        self.response_repo = ResponseRepository(db)
        self.applicant_repo = ApplicantRepository(db)
        self.violation_repo = ViolationRepository(db)
        self.job_model_repo = JobModelRepository(db)

    def grade_answer(self, question, answer_data):
        q_data = json.loads(question.question_data)
        points = question.points

        if question.q_type == "mcq":
            selected = answer_data.get("selected_index") if answer_data else None
            correct = q_data.get("correct_index")
            try:
                is_correct = selected is not None and int(selected) == int(correct)
            except (TypeError, ValueError):
                is_correct = False
            return is_correct, (points if is_correct else 0)

        if question.q_type in ("excel_formula", "code_correction", "language_check"):
            raw_answer = answer_data.get("text", "") if answer_data else ""
            normalized_answer = normalize_text(raw_answer)
            keywords = q_data.get("answer_keywords", [])
            is_correct = any(
                normalize_text(kw) and normalize_text(kw) in normalized_answer for kw in keywords
            )
            return is_correct, (points if is_correct else 0)

        return False, 0

    def save_and_grade(self, user_id: str, question_id: str, answer_data):
        question = self.question_repo.get(question_id)
        if not question:
            raise ValueError("السؤال غير موجود")

        is_correct, score = self.grade_answer(question, answer_data)
        stored_answer = json.dumps(answer_data, ensure_ascii=False) if answer_data is not None else None
        self.response_repo.upsert(user_id, question_id, stored_answer, is_correct, score)
        return {"is_correct": is_correct, "score": score}

    def finalize_exam(self, user_id: str):
        applicant = self.applicant_repo.get(user_id)
        if not applicant:
            raise ValueError("المتقدم غير موجود")

        questions = self.question_repo.list_by_job_model(applicant.job_model_id)
        responses = {r.question_id: r for r in self.response_repo.list_by_user(user_id)}

        total_score = 0.0
        max_score = 0.0
        details = []
        for q in questions:
            max_score += q.points
            resp = responses.get(q.id)
            score = resp.score if resp else 0
            is_correct = bool(resp.is_correct) if resp else False
            total_score += score
            details.append({
                "question_id": q.id,
                "question_text": q.question_text,
                "q_type": q.q_type,
                "points": q.points,
                "score": score,
                "is_correct": is_correct,
                "answer": json.loads(resp.answer_data) if resp and resp.answer_data else None,
            })

        applicant.status = "finished"
        applicant.exam_finished_at = datetime.utcnow()
        applicant.total_score = total_score
        applicant.max_score = max_score
        self.db.commit()

        return {"total_score": total_score, "max_score": max_score, "details": details}

    def get_result(self, user_id: str):
        applicant = self.applicant_repo.get(user_id)
        if not applicant:
            raise ValueError("المتقدم غير موجود")

        job_model = self.job_model_repo.get(applicant.job_model_id)
        questions = self.question_repo.list_by_job_model(applicant.job_model_id)
        responses = {r.question_id: r for r in self.response_repo.list_by_user(user_id)}

        details = []
        for q in questions:
            resp = responses.get(q.id)
            details.append({
                "question_id": q.id,
                "question_text": q.question_text,
                "q_type": q.q_type,
                "points": q.points,
                "score": resp.score if resp else 0,
                "is_correct": bool(resp.is_correct) if resp else False,
                "answer": json.loads(resp.answer_data) if resp and resp.answer_data else None,
            })

        violations = self.violation_repo.list_by_user(user_id)

        return {
            "applicant": applicant,
            "job_model_name": job_model.name if job_model else "",
            "details": details,
            "violations": violations,
        }

"""
معالجة المزامنة اللحظية لإجابات المتقدمين، مع إعادة المحاولة تلقائياً
عند حدوث قفل مؤقت لقاعدة البيانات نتيجة الكتابة المتزامنة من عدد كبير
من المتقدمين في نفس اللحظة.
"""

import time

from sqlalchemy.exc import OperationalError

from services.exam_service import ExamService


def process_single_answer(db, user_id: str, question_id: str, answer_data, max_retries: int = 3):
    attempt = 0
    last_error = None
    while attempt < max_retries:
        try:
            result = ExamService(db).save_and_grade(user_id, question_id, answer_data)
            return {"success": True, "question_id": question_id, "result": result}
        except OperationalError as e:
            db.rollback()
            last_error = str(e)
            attempt += 1
            time.sleep(0.3 * attempt)
        except ValueError as e:
            return {"success": False, "question_id": question_id, "error": str(e)}
    return {"success": False, "question_id": question_id, "error": last_error or "فشل حفظ الإجابة بعد عدة محاولات"}


def process_batch(db, user_id: str, answers):
    """answers: قائمة عناصر AnswerPayload تحتوي question_id و answer_data."""
    results = []
    for item in answers:
        results.append(process_single_answer(db, user_id, item.question_id, item.answer_data))
    return results
